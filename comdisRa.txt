**Task 09: NFS Server + AutoFS Direct Map [alpha & beta]**

```bash

sudo systemctl restart autofs
```

**โจทย์ข้อที่ 21: การจัดการ Rootless Container และ Quadlet สำหรับ User**

```bash
# 1. รันในสิทธิ์ root
sudo useradd user80
echo "user80:redhat" | sudo chpasswd
sudo mkdir -p /data01
sudo chown user80:user80 /data01

# 2. สลับเป็น user80
su - user80
mkdir -p ~/.config/containers/systemd
cat > ~/.config/containers/systemd/c80.container <<'EOF'
[Unit]
Description=User80 Rootless Container

[Container]
ContainerName=c80
Image=registry.access.redhat.com/ubi9/ubi:latest
Exec=sleep infinity
Volume=/data01:/data01:Z

[Install]
WantedBy=default.target
EOF

systemctl --user daemon-reload
systemctl --user enable --now c80.service
touch /data01/testfile_from_host
podman exec c80 ls -l /data01
```
